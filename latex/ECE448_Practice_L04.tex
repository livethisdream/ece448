\question \textbf{Choosing a radio.} The course SDRs are the RTL-SDR Blog V4
(0.5 to 1766 MHz, 2.56 MS/s, 8 bits, receive only), the HackRF One (1 to 6000 MHz,
20 MS/s, 8 bits, half duplex), the PlutoSDR (325 to 3800 MHz, 20 MHz analog bandwidth,
12 bits, full duplex), and the ADALM-PHASER (10.0 to 10.5 GHz, digitized by a PlutoSDR).
\begin{parts}

	\part For each task, name a course SDR that can do it, and give the specification
	that rules out each of the others.
	\begin{enumerate}[label=(\roman*)]
		\item Receive ship AIS at 162 MHz.
		\item Display the whole FM broadcast band, 88 to 108 MHz, at once.
		\item Transmit a 915 MHz test signal into a cable and receive it at the same time.
		\item Find the direction of a 10.3 GHz beacon.
	\end{enumerate}
		\begin{solution}[2.3in]
		(i) The RTL-SDR or the HackRF; 162 MHz is inside both tuning ranges. The
		PlutoSDR starts at 325 MHz and the ADALM-PHASER works only at 10.0 to 10.5 GHz.

		(ii) The HackRF at 20 MS/s, which shows a 20 MHz window. The RTL-SDR's
		2.56 MS/s shows only 2.56 MHz, the PlutoSDR cannot tune to 100 MHz, and the
		ADALM-PHASER works only at X band.

		(iii) The PlutoSDR, the only full-duplex radio. The HackRF transmits or receives
		but not both at once, the RTL-SDR cannot transmit, and the ADALM-PHASER does
		not cover 915 MHz.

		(iv) The ADALM-PHASER: 10.3 GHz is inside its band, its 8 elements let it find
		direction, and every other radio stops at 6 GHz or below.
		\end{solution}

	\part Explain how an SDR differs from a hardware FM radio, and why the same
	RTL-SDR can decode both broadcast FM and aircraft ADS-B.
		\begin{solution}[1.1in]
		A hardware radio builds its filters and demodulator as circuits for one kind of
		signal. An SDR only tunes, filters, and digitizes a band, and software does the
		filtering, demodulation, and decoding. Both 100 MHz and 1090 MHz are inside the
		RTL-SDR's tuning range, so receiving either one is a matter of setting the
		center frequency and running a different flowgraph.
		\end{solution}

\end{parts}

\question \textbf{The mixer and I and Q.} An RTL-SDR's center frequency, its LO, is
set to 433.0 MHz, and a key fob transmits at 433.92 MHz.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the two frequencies at the mixer output, and say which one the
		low-pass filter keeps.
		\begin{solution}[1.0in]
		\eq{ 433.92 - 433.0 &= 0.92\ \text{MHz} \\ 433.92 + 433.0 &= 866.92\ \text{MHz} }
		The low-pass filter keeps the 0.92 MHz difference and removes the sum.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		kept: \ansbox[1.2in]{0.92 MHz}

		\vspace{4pt}
		removed: \ansbox[1.2in]{866.92 MHz}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part With a single real mixer, a second transmitter would also come out at
		0.92 MHz. Find its frequency, and explain how the SDR's $I$ and $Q$ outputs tell
		the two apart.
		\begin{solution}[1.3in]
		\eq{ 433.0 - 0.92 = 432.08\ \text{MHz} }
		The second mixer, driven by the LO shifted 90°, gives $Q$. In $I + jQ$ the
		433.92 MHz signal rotates one way and appears at $+0.92$ MHz, and the
		432.08 MHz signal rotates the other way and appears at $-0.92$ MHz.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.2in]{432.08 MHz}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part A superheterodyne receiver, like the ADALM-PHASER, mixes a 10.25 GHz
		signal to an IF of 2.2 GHz with an LO above the signal. Find the LO frequency,
		and the other input frequency that also mixes to 2.2 GHz.
		\begin{solution}[1.3in]
		\eq{ f_{LO} &= 10.25 + 2.2 = 12.45\ \text{GHz} \\ f_\text{image} &= 12.45 + 2.2 = 14.65\ \text{GHz} }
		A filter ahead of the mixer must remove 14.65 GHz, or a signal there lands on
		top of the wanted one at the IF.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		LO: \ansbox[1.2in]{12.45 GHz}

		\vspace{4pt}
		other: \ansbox[1.2in]{14.65 GHz}
	\end{minipage}

\end{parts}

\question \textbf{Sample rate and data rate.} A team captures one 20 MHz Wi-Fi channel
centered on 2437 MHz with a HackRF.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the lowest sample rate that shows the whole channel, and the band the
		window covers.
		\begin{solution}[0.9in]
		Complex samples at $f_s$ show a window $f_s$ wide, so $f_s = 20$ MS/s:
		\eq{ 2437 \pm 10 = 2427 \text{ to } 2447\ \text{MHz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		rate: \ansbox[1.3in]{20 MS/s}

		\vspace{4pt}
		band: \ansbox[1.3in]{2427 to 2447 MHz}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part GNU Radio stores each complex sample in 8 bytes. Find the data rate
		through the flowgraph and the size of a 10 s File Sink capture.
		\begin{solution}[0.9in]
		\eq{ 8 \times 20\times10^{6} &= 160\ \text{MB/s} \\ 10 \times 160 &= 1.6\ \text{GB} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		rate: \ansbox[1.2in]{160 MB/s}

		\vspace{4pt}
		file: \ansbox[1.2in]{1.6 GB}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part The HackRF sends 8-bit $I$ and 8-bit $Q$ over USB. Find the USB data rate
		at 20 MS/s, and how many times larger the flowgraph's rate is.
		\begin{solution}[0.9in]
		\eq{ 2\ \text{bytes} \times 20\times10^{6} = 40\ \text{MB/s}, \qquad \frac{160}{40} = 4 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		USB: \ansbox[1.2in]{40 MB/s}

		\vspace{4pt}
		ratio: \ansbox[1.2in]{4}
	\end{minipage}

	\part A teammate suggests the RTL-SDR instead. Give two separate reasons it
	cannot make this capture.
		\begin{solution}[0.9in]
		Its tuning range stops at 1766 MHz, below 2437 MHz. Even inside its range, its
		2.56 MS/s shows only 2.56 MHz, far less than the 20 MHz channel.
		\end{solution}

\end{parts}

\question \textbf{Bits and dynamic range.} Each ADC bit adds about 6 dB of dynamic
range. A strong local station reaches the ADC at $-10$ dBFS, and a weak station of
interest is 55 dB below it.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Estimate the dynamic range of an 8-bit and a 12-bit converter, and the
		level of the weak station in dBFS.
		\begin{solution}[0.9in]
		\eq{ 8 \times 6 &= 48\db, \qquad 12 \times 6 = 72\db \\ -10 - 55 &= -65\ \text{dBFS} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		8 / 12 bits: \ansbox[1.2in]{48 / 72 dB}

		\vspace{4pt}
		weak: \ansbox[1.2in]{$-65$ dBFS}
	\end{minipage}

	\part The converter's own floor sits about one dynamic range below full scale.
	Which converter shows the weak station, and by how many dB does it clear that
	converter's floor?
		\begin{solution}[0.9in]
		The 8-bit floor is near $-48$ dBFS, above the $-65$ dBFS station, so the 8-bit
		radio does not show it. The 12-bit floor is near $-72$ dBFS, so the 12-bit radio
		shows it about $-65 - (-72) = 7$ dB above the floor.
		\end{solution}

	\part On the 8-bit radio, how much can the gain rise before the strong station
	clips, and does that rise make the weak station visible? Explain.
		\begin{solution}[1.2in]
		The strong station is 10 dB below full scale, so the gain can rise 10 dB. The
		weak station rises by the same 10 dB, to $-55$ dBFS, which is still below the
		$-48$ dBFS floor. Gain moves both stations together and cannot widen the 48 dB
		span between full scale and the floor; seeing both needs more bits or a filter
		that keeps the strong station out of the window.
		\end{solution}

	\part In the L4 flowgraph, name two things on the plots that show the ADC is
	clipping.
		\begin{solution}[0.8in]
		The time sink shows $I$ and $Q$ flattened at $\pm 1$, and the spectrum shows new
		signals where nothing was transmitted. The strong tone also rises by less than
		the gain increase.
		\end{solution}

\end{parts}
