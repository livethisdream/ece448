\question \textbf{Planning a capture.} The course radio is the PlutoSDR: 70 to 6000 MHz
in the extended-frequency mode the course Plutos use, about 4 MS/s sustained over USB
and up to 61.44 MS/s in short bursts, 12 bits, full duplex.
\begin{parts}

	\part For each task, say whether the PlutoSDR can do it and how, or name the
	specification that rules it out.
	\begin{enumerate}[label=(\roman*)]
		\item Receive ship AIS at 162 MHz.
		\item Display the whole FM broadcast band, 88 to 108 MHz, at once.
		\item Transmit a 915 MHz test signal into a cable, through a 30 dB attenuator, and receive it at the same time.
		\item Find the direction of a 10.3 GHz beacon.
	\end{enumerate}
		\begin{solution}[2.3in]
		(i) Yes. 162 MHz is inside the extended-frequency range, 70 to 6000 MHz; a
		stock Pluto, which starts at 325 MHz, could not tune it.

		(ii) Not while streaming: its 4 MS/s shows only 4 MHz of the 20 MHz band. A
		short burst capture at 20 MS/s or more shows the whole band at once, and a
		rate above 20 MS/s leaves margin for the filters' roll-off at the edges.

		(iii) Yes. The PlutoSDR is full duplex, so its transmit port is cabled through
		the 30 dB attenuator to its receive port and both run at once.

		(iv) No. 10.3 GHz is above the top of its tuning range, 6 GHz.
		\end{solution}

	\part Explain how an SDR differs from a hardware FM radio, and why the same
	PlutoSDR can decode both broadcast FM and aircraft ADS-B.
		\begin{solution}[1.1in]
		A hardware radio builds its filters and demodulator as circuits for one kind of
		signal. An SDR only tunes, filters, and digitizes a band, and software does the
		filtering, demodulation, and decoding. Both 100 MHz and 1090 MHz are inside the
		PlutoSDR's extended-frequency tuning range, so receiving either one is a matter of setting the
		center frequency and running a different flowgraph.
		\end{solution}

\end{parts}

\question \textbf{The mixer and I and Q.} We set a PlutoSDR's center frequency, its LO,
to 433.0 MHz, and a key fob transmits at 433.92 MHz.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the two frequencies at the mixer output, and say which one the
		low-pass filter keeps.
		\begin{solution}[1.0in]
		\eq{ 433.92 - 433.0 &= 0.92\ \text{MHz} \\ 433.92 + 433.0 &= 866.92\ \text{MHz} }
		The low-pass filter keeps the 0.92 MHz difference and removes the sum.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		kept: \ansbox[1.2in]{0.92 MHz}

		\vspace{4pt}
		removed: \ansbox[1.2in]{866.92 MHz}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part With a single real mixer, a second transmitter would also come out at
		0.92 MHz. Find its frequency, and explain how the PlutoSDR's $I$ and $Q$ outputs
		tell the two apart.
		\begin{solution}[1.3in]
		\eq{ 433.0 - 0.92 = 432.08\ \text{MHz} }
		The second mixer, driven by the LO shifted 90°, gives $Q$. In $I + jQ$ the
		433.92 MHz signal rotates one way and appears at $+0.92$ MHz, and the
		432.08 MHz signal rotates the other way and appears at $-0.92$ MHz.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.2in]{432.08 MHz}
	\end{minipage}

\end{parts}

\question \textbf{Sample rate and data rate.} A team captures a 10 MHz-wide test signal
centered on 2437 MHz with a PlutoSDR, in a short burst. The usable window is somewhat
narrower than $f_s$, because the radio's filters roll off near its edges.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the lowest sample rate whose window just holds the signal. Then
		choose a rate 25\% higher, to leave margin at the edges, and find the band
		that window covers.
		\begin{solution}[1.1in]
		A complex window is $f_s$ wide, so the signal needs at least 10 MS/s. With
		25\% margin:
		\eq{ f_s &= 1.25 \times 10 = 12.5\ \text{MS/s} \\ 2437 \pm 6.25 &= 2430.75 \text{ to } 2443.25\ \text{MHz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		lowest: \ansbox[1.3in]{10 MS/s}

		\vspace{4pt}
		chosen: \ansbox[1.3in]{12.5 MS/s}

		\vspace{4pt}
		band: \ansbox[1.6in]{2430.75 to 2443.25 MHz}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part GNU Radio stores each complex sample in 8 bytes. At 12.5 MS/s, find the
		data rate through the flowgraph and the size of a 10 s File Sink capture.
		\begin{solution}[0.9in]
		\eq{ 8 \times 12.5\times10^{6} &= 100\ \text{MB/s} \\ 10 \times 100 &= 1\ \text{GB} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		rate: \ansbox[1.2in]{100 MB/s}

		\vspace{4pt}
		file: \ansbox[1.2in]{1 GB}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part The PlutoSDR sends each 12-bit $I$ and $Q$ over USB as a 16-bit word.
		Find the USB data rate at 12.5 MS/s, and compare it with the rate at the
		4 MS/s the Pluto streams without dropped samples.
		\begin{solution}[0.9in]
		\eq{ 4\ \text{bytes} \times 12.5\times10^{6} &= 50\ \text{MB/s} \\ 4\ \text{bytes} \times 4\times10^{6} &= 16\ \text{MB/s} }
		The capture needs about three times what the Pluto streams over its USB 2.0 link, so it
		works only as a short burst into the Pluto's memory.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		USB: \ansbox[1.2in]{50 MB/s}

		\vspace{4pt}
		at 4 MS/s: \ansbox[1.2in]{16 MB/s}
	\end{minipage}

	\part A teammate suggests streaming the capture live at 12.5 MS/s instead of taking
	a burst. Explain what goes wrong.
		\begin{solution}[0.9in]
		The Pluto streams only about 4 MS/s over its USB 2.0 link without dropping
		samples. At 12.5 MS/s it drops most of them, so the recording has gaps and the
		spectrum shows false signals and a raised floor.
		\end{solution}

\end{parts}

\question \textbf{Bits and clipping.} Use the L4 flowgraph, \texttt{L04-adc-bits.grc},
and its plots on the lesson page.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part How many levels does the ADC block round to with 8 bits, the resolution
		of an 8-bit radio, and with 12 bits, the PlutoSDR's?
		\begin{solution}[0.8in]
		\eq{ 2^{8} = 256, \qquad 2^{12} = 4096 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		8 bits: \ansbox[1.2in]{256}

		\vspace{4pt}
		12 bits: \ansbox[1.2in]{4096}
	\end{minipage}

	\part At 8 bits the weak tone at $-0.49$ MHz is not visible. Switching the menu to
	12 bits makes it appear, about 40 dB above the floor. Did the weak signal change?
	Describe what changed on the spectrum, and why.
		\begin{solution}[1.1in]
		The weak signal did not change; the input is the same. The floor dropped by about
		20 dB and the spurs disappeared, because 12 bits round in much finer steps, so the
		rounding error is smaller. The weak tone was always there, below the 8-bit floor.
		\end{solution}

	\part Set the flowgraph back to 8 bits. Raise the gain slider in steps until the
	time sink's $I$ and $Q$ just begin to flatten at full scale. Did the weak tone
	appear on the spectrum? Explain what you saw, and name one thing that would let an
	8-bit radio show the weak signal.
		\begin{solution}[1.3in]
		No. Gain raised the strong tone, the weak tone, and the spurs together, so the
		weak tone stayed hidden below the spurs, and the strong tone reached full scale
		first. Raising the gain further only clipped the strong tone. More bits would
		help, or a filter that keeps the strong signal out of the ADC.
		\end{solution}

	\part In the L4 flowgraph, name two things on the plots that show the ADC is
	clipping.
		\begin{solution}[0.8in]
		The time sink shows $I$ and $Q$ flattened at $\pm 1$, and the spectrum shows new
		signals where nothing transmits. The strong tone also rises by less than
		the gain increase.
		\end{solution}

\end{parts}
