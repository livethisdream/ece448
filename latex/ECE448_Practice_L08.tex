\question \textbf{Transmitter settings.} A flowgraph samples at $f_s = 48$ kHz and
sends FSK at 1200 baud with a deviation of $\Delta f = 600$ Hz, one bit per symbol.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the samples per symbol to enter in the Repeat block.
		\begin{solution}[0.7in]
		\eq{ \text{sps} = \dfrac{f_s}{R_s} = \dfrac{48000}{1200} = 40 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{40}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the sensitivity to enter in the Frequency Mod block.
		\begin{solution}[0.8in]
		\eq{ k = \dfrac{2\pi \Delta f}{f_s} = \dfrac{2\pi \times 600}{48000} = 0.0785\ \text{rad/sample} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{0.0785}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the modulation index and estimate the bandwidth with Carson's rule.
		\begin{solution}[1.0in]
		\eq{ h &= \dfrac{2\Delta f}{R_s} = \dfrac{1200}{1200} = 1 \\
			B &\approx 2(\Delta f + R_s) = 2(600 + 1200) = 3.6\ \text{kHz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$h$ = \ansbox[1.0in]{1}

		\vspace{4pt}
		$B$ = \ansbox[1.0in]{3.6 kHz}
	\end{minipage}

	\part A classmate builds the same transmitter from two Signal Sources at $+600$
	and $-600$ Hz and a selector that switches between them at each bit. Explain why
	Frequency Mod gives a more compact spectrum.
		\begin{solution}[1.0in]
		The two sources run independently, so at each switch the phase jumps from
		wherever one source is to wherever the other is. Those jumps are abrupt
		changes in the signal, and abrupt changes spread power across a wide band.
		Frequency Mod adds a phase step every sample, so the phase is continuous
		across each bit boundary and the spectrum stays compact.
		\end{solution}

\end{parts}

\question \textbf{Receiver settings.} The receiver for the link of Question 1 follows
the L8 flowgraph: Low Pass Filter, Quadrature Demod, Moving Average, Symbol Sync,
and Binary Slicer.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the Quadrature Demod gain that puts the two tones at $+1$ and $-1$.
		\begin{solution}[0.8in]
		\eq{ g = \dfrac{f_s}{2\pi\Delta f} = \dfrac{48000}{2\pi \times 600} = 12.7 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{12.7}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the low-pass filter cutoff that passes the Carson bandwidth.
		\begin{solution}[0.7in]
		The filter passes $-f_c$ to $+f_c$, so the cutoff is half of $B$:
		\eq{ f_c = \Delta f + R_s = 600 + 1200 = 1800\ \text{Hz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{1800 Hz}
	\end{minipage}

	\part A classmate leaves the Quadrature Demod gain at 1. Find the levels the two
	tones produce, and explain whether the slicer still decides correctly.
		\begin{solution}[1.1in]
		With $g = 1$ the output is the phase step itself,
		$\pm 2\pi\Delta f/f_s = \pm 0.0785$. The slicer compares with zero, and both
		levels are still on the correct side of zero, so the bits come out correctly.
		Only the scale is wrong: the eye diagram and any plot set for $\pm 1$ show a
		nearly flat line.
		\end{solution}

	\part The classmate also sets the filter cutoff to 8 kHz. Explain what changes
	on the eye diagram, and what happens to the noise voltage at which the text first
	breaks.
		\begin{solution}[1.0in]
		The signal still fits through the filter, but the filter now passes far more
		noise, everything from $-8$ to $+8$ kHz instead of $-1.8$ to $+1.8$ kHz. The eye's
		traces are thicker at the same noise setting, and the text breaks at a lower
		noise voltage than with the 1800 Hz cutoff.
		\end{solution}

\end{parts}

\question \textbf{Noise.} The link of Questions 1 and 2 ($f_s = 48$ kHz, 1200 baud,
$\Delta f = 600$ Hz, cutoff 1800 Hz) runs through a Channel Model with a noise voltage
slider. With the slider low, the eye is wide open and every line of text arrives.
\begin{parts}

	\part The noise slider is raised a long way. Describe what you expect to see on
	the spectrum, on the eye diagram, and in the received text.
		\begin{solution}[1.1in]
		The tones stay where they are and the floor rises toward them. The eye's
		traces thicken, and now and then a sharp spike, an FM click, crosses zero.
		Lines start to arrive with wrong characters, and with still more noise the
		receiver stops finding the access code.
		\end{solution}

	\part A spike crosses zero on the eye diagram, but away from the sampling instant.
	Explain whether it causes a bit error.
		\begin{solution}[0.9in]
		No. The Binary Slicer only sees the one sample per bit that Symbol Sync picks
		at the peak of each bit. A spike that crosses zero between those instants never
		reaches the slicer; only one at a sampling instant flips a bit.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part The link raises its symbol rate to 1600 baud and keeps $\Delta f = 600$ Hz.
		Find the new samples per symbol, bandwidth, and filter cutoff.
		\begin{solution}[1.0in]
		\eq{ \text{sps} &= 48000/1600 = 30 \\
			B &= 2(600 + 1600) = 4.4\ \text{kHz}, \qquad f_c = 2.2\ \text{kHz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		sps = \ansbox[1.0in]{30}

		\vspace{4pt}
		$B$ = \ansbox[1.0in]{4.4 kHz}

		\vspace{4pt}
		$f_c$ = \ansbox[1.0in]{2.2 kHz}
	\end{minipage}

	\part Explain whether the 1600-baud link's text breaks at a higher or a lower
	noise setting than the 1200-baud link's.
		\begin{solution}[0.8in]
		Lower. The wider filter lets in more of the noise, so the eye is noisier at the
		same slider setting and the text breaks sooner.
		\end{solution}

	\part On an FSK link the text goes from clean to almost no packets over a small
	range of the noise slider. Lesson 2 designs a link with margin. Explain, from that
	observation, why margin matters for FSK.
		\begin{solution}[1.0in]
		A link that only just works at its design distance fails completely when the
		signal fades a little. Margin keeps the link on the clean side when the path
		loss changes.
		\end{solution}

\end{parts}

\question \textbf{An FSK remote on a spectrogram.} An RTL-SDR capture, viewed as a
spectrogram in URH, shows a burst with two tracks, at 433.900 and 433.940 MHz. In the
alternating preamble, the signal stays on each track for 0.417 ms.
\begin{parts}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the center frequency, the deviation, and the symbol rate.
		\begin{solution}[1.1in]
		\eq{ f_c &= \dfrac{433.900 + 433.940}{2} = 433.920\ \text{MHz} \\
			\Delta f &= \dfrac{40\ \text{kHz}}{2} = 20\ \text{kHz} \\
			R_s &= \dfrac{1}{0.417\ \text{ms}} = 2400\ \text{baud} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$f_c$ = \ansbox[1.2in]{433.920 MHz}

		\vspace{4pt}
		$\Delta f$ = \ansbox[1.2in]{20 kHz}

		\vspace{4pt}
		$R_s$ = \ansbox[1.2in]{2400 baud}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part The remote's crystal is accurate to 10 ppm. Find the largest frequency
		offset at 433.92 MHz.
		\begin{solution}[0.7in]
		\eq{ 10\times10^{-6} \times 433.92\times10^{6} = 4.3\ \text{kHz} }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{4.3 kHz}
	\end{minipage}

	\part An offset adds $f_\text{off}/\Delta f$ to both demodulator levels. Find the
	remote's two levels with a 4.3 kHz offset, and say whether the slicer still decides
	correctly. Then do the same for a 433.92 MHz link built with $\Delta f = 2$ kHz.
		\begin{solution}[1.3in]
		Remote: $4.3/20 = 0.22$, so the levels are $+1.22$ and $-0.78$. Both are on
		the correct side of zero, so the slicer still decides correctly; the large
		deviation is what makes a 10 ppm crystal acceptable. With $\Delta f = 2$ kHz:
		$4.3/2 = 2.15$, so the levels would be $+3.15$ and $+1.15$, both above zero, and
		every 0 would come out as a 1. In practice the tones also move off the
		receiver's filter, and the link fails.
		\end{solution}

	\part Name one change to the remote's receiver that would put its levels back at
	$\pm 1$.
		\begin{solution}[0.9in]
		Subtract the average of the demodulator output, for example with a DC Blocker
		after the Quadrature Demod, because the offset appears as a constant
		$f_\text{off}/\Delta f$ added to both levels. Measuring the offset and shifting
		the signal back in frequency before the filter also works.
		\end{solution}

\end{parts}
