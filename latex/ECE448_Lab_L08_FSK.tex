\begin{parts}

	\begin{tabular}{| l | K{0.82in} | K{0.82in} | K{0.82in} | K{0.82in} | K{0.82in} |}
	\hline
	$\Delta f$ & sensitivity $k$ & demod gain $g$ & index $h$ & Carson $B$ & LPF cutoff \\\hline

	\part With $\Delta f = 1$ kHz, predict the SNR at the demodulator for each noise
	voltage. The signal has power 1, and noise of voltage $\sigma$ has power
	$\sigma^2$ over the full 32 kHz.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.6in} | K{0.6in} | K{0.6in} | K{0.6in} | K{0.6in} | K{0.6in} |}
	\hline
	$\sigma$ & $0.5$ & $0.8$ & $1.0$ & $1.2$ & $1.5$ & $2.0$ \\\hline
	SNR (dB) & \kv{$15.1$} & \kv{$11.0$} & \kv{$9.0$} & \kv{$7.4$} & \kv{$5.5$} & \kv{$3.0$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.6in]
		The filter keeps $4/32$ of the noise, so
		$\text{SNR} = 1/(\sigma^2 \times 4/32) = 8/\sigma^2$.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part Each line is a packet: a 64-bit access code, a 32-bit header, and the 41

		\eq{ 64 + 32 + 8 \times 41 &= 424\ \text{bits} \\

\fullwidth{\large\textbf{Part 3: Writeup} \normalsize(individual, after the lab)}

\question \textbf{Analysis.}

\begin{parts}

	\part Find the SNR from Question 2(b) at which your link first lost a line, and
	the SNR at which it received almost nothing. Explain why the change takes only a
	few dB, and what that means for the link margin of Lesson 2.
		\begin{solution}[1.5in]
		Errors start near 9 dB ($\sigma \approx 1.0$) and packets stop near 3 dB
		($\sigma = 2.0$), about 6 dB apart. Below about 10 dB the noise begins to swing
		the phase of the received signal by large amounts, and each large swing makes
		a spike in the demodulator output; the number of spikes grows rapidly as the
		SNR falls. A packet also needs every one of its 424 bits right, so a small rise
		in bit errors loses many packets. A link designed at the threshold fails on the
		first fade of a few dB, so it needs margin above 9 dB.
		\end{solution}

	\part Your link tolerated an offset of about 600 Hz. Convert that to parts per
	million at 915 MHz, and explain what it says about running this link over the air
	between two SDRs whose crystals are accurate to a few ppm.
		\begin{solution}[1.3in]
		\eq{ \dfrac{600}{915\times10^{6}} = 0.66\ \text{ppm} }
		Two radios with a few ppm of error each can be several kHz apart at 915 MHz, far
		more than 600 Hz. Over the air this link would need either a much larger
		deviation, as key fobs use, or a receiver that measures the offset and removes
		it before the demodulator.
		\end{solution}

	\part Recommend a deviation of 1000 or 2000 Hz for this link, and justify it from
	your measurements in Questions 4 and 5. Name what the choice gives up.
		\begin{solution}[1.3in]
		2000 Hz: it tolerates twice the offset and did slightly better at
		$\sigma = 1.2$. It gives up bandwidth, 6 kHz instead of 4 kHz, so fewer such
		links fit in a band. A cadet who argues for 1000 Hz because the link is
		cabled and the bandwidth is scarce, with the measurements cited, also meets the
		objective.
		\end{solution}

\end{parts}
