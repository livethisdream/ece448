\question \textbf{Settings and readings.} Question 3 checks the bandwidth against your spectrum.

\begin{parts}

	\part Fill in the transmitter and receiver settings for each deviation.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.95in} | K{0.95in} | K{0.95in} | K{0.95in} |}
	\hline
	$\Delta f$ & sensitivity $k$ & demod gain $g$ & bandwidth $B$ & LPF cutoff \\\hline
	1500 Hz & \kv{0.295} & \kv{3.40} & \kv{5 kHz} & \kv{2.5 kHz} \\\hline
	2000 Hz & \kv{0.393} & \kv{2.55} & \kv{6 kHz} & \kv{3 kHz} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.9in]
		$k = 2\pi\Delta f/f_s$, $g = f_s/(2\pi\Delta f)$, $B = 2(\Delta f + R_s)$
		with $R_s = 32000/32 = 1000$ baud, and the cutoff is $\Delta f + R_s$, half
		of $B$.
		\end{solution}

	\part Run \texttt{L08-fsk-link.grc}. On its spectrum, read how far the two tones
	stand above the noise floor with the noise slider at 0.5, then at 1.0. Describe
	what the eye diagram does between the two.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{1.4in} | K{1.4in} |}
	\hline
	Noise voltage & $0.5$ & $1.0$ \\\hline
	Tones above floor (dB) & \kv{$\approx 25$} & \kv{$\approx 19$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.8in]
		The floor rose about 6 dB from 0.5 to 1.0; the tones stayed where they
		were. The eye's traces thicken but the eye stays open. Readings within a few
		dB are fine, since the floor on the plot is ragged.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part Each line is a packet: a 64-bit access code, a 32-bit header, and the 41
		bytes of the line, its 40 characters and a newline. Find the bits per packet, the time per packet, and
		the number of packets in 30 s.
		\begin{solution}[1.0in]
		\eq{ 64 + 32 + 8 \times 41 &= 424\ \text{bits} \\
			424 / 1000 &= 0.424\ \text{s} \\
			30 / 0.424 &\approx 70.8 \Rightarrow 70 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		bits = \ansbox[1.0in]{424}

		\vspace{4pt}
		time = \ansbox[1.0in]{0.424 s}

		\vspace{4pt}
		in 30 s = \ansbox[1.0in]{70}
	\end{minipage}


\end{parts}

\fullwidth{\large\textbf{Part 3: Writeup} \normalsize(individual, after the lab)}

\question \textbf{Analysis.}

\begin{parts}

	\part From your table in Question 4 and your screenshots, describe in words what
	the eye and the received text did as you raised the noise from 0.5 to 2.0. Explain
	why a small turn of the slider takes the link from clean text to no text, and what
	that means for the link margin of Lesson 2.
		\begin{solution}[1.5in]
		Clean text and a wide-open eye up to about 1.0; then the eye thickens and
		sharp spikes, FM clicks, begin to cross zero, and lines start to have errors;
		by 2.0 almost no packets arrive. Once the noise is large enough to swing the
		phase by a lot, each swing makes a spike, and the spikes grow quickly with a
		little more noise. A packet also needs every one of its 424 bits right, so a
		few bit errors lose many packets. A link that only just works fails on the
		first fade, so it needs margin.
		\end{solution}

	\part Take the largest offset at which your link still delivered correct lines in
	Question 5(a). Convert it to parts per
	million at 915 MHz, and explain what it says about running this link over the air
	between two SDRs whose crystals are accurate to a few ppm.
		\begin{solution}[1.3in]
		\eq{ \dfrac{600}{915\times10^{6}} = 0.66\ \text{ppm} }
		Two radios with a few ppm of error each can be several kHz apart at 915 MHz, far
		more than 600 Hz. Over the air this link would need either a much larger
		deviation, as key fobs use, or a receiver that measures the offset and removes
		it before the demodulator.
		\end{solution}

	\part Recommend a deviation of 1000 or 2000 Hz for this link, and justify it from
	your measurements in Questions 4 and 5. Name what the choice gives up.
		\begin{solution}[1.3in]
		2000 Hz: it tolerates twice the offset and did slightly better at
		$\sigma = 1.2$. It gives up bandwidth, 6 kHz instead of 4 kHz, so fewer such
		links fit in a band. A cadet who argues for 1000 Hz because the link is
		cabled and the bandwidth is scarce, with the measurements cited, also meets the
		objective.
		\end{solution}

\end{parts}
