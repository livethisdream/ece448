\begin{parts}

	capture. The shift is the Frequency Xlating FIR Filter's center frequency.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}

	Recording & SDR center (MHz) & Station (MHz) & Shift (kHz) \\\hline

	\end{minipage}

\question \textbf{Build the WBFM receiver.} Start GRC with an empty canvas. Do not
open \texttt{L07-wbfm-rx.grc}: build the receiver yourself.
\begin{itemize}\setlength{\itemsep}{0pt}
	\item \texttt{samp\_rate} 2.4e6, a QT GUI Entry \texttt{fc\_mhz} (100.0), a QT GUI
	Entry \texttt{station\_mhz} (100.3), and a QT GUI Range \texttt{vol} from 0 to 1.
	\item A File Source on \texttt{L07-fm-test.cfile}, repeat on.
	\item A Frequency Xlating FIR Filter, decimation 10, center frequency
	\texttt{(station\_mhz - fc\_mhz) * 1e6}, with a Low-pass Filter Taps variable
	\texttt{chan\_taps}: cutoff 105 kHz, transition 30 kHz, at \texttt{samp\_rate}.
	\item A Quadrature Demod with the gain of Question 1(b).
	\item A Decimating FIR Filter, decimation 5, with taps
	\texttt{firdes.low\_pass(1, samp\_rate/10, 15e3, 3e3)}.
	\item FM Deemphasis at 48 kHz with $\tau$ = 75~$\mu$s, a Multiply Const
	\texttt{vol}, an Audio Sink at 48 kHz, and a float QT GUI Frequency Sink on the
	de-emphasis output. Do not add a Throttle.
\end{itemize}

\begin{parts}

	\part Paste a screenshot of your flowgraph on the GRC canvas. Every connection
	must be black, not red.
		\begin{solution}[1.6in]
		Matches the L7 lesson figure: File Source, Frequency Xlating FIR Filter (complex
		ports), Quadrature Demod, then float blocks: the decimating audio filter, FM
		Deemphasis, Multiply Const, and the Audio Sink.
		\end{solution}

	\part Run it. Read the audio frequency sink, and record the frequency of each line and how far the
	2.5~kHz line sits below the 1~kHz line.
		\begin{solution}[0.6in]
		The sink shows lines at 1 and 2.5 kHz, about 3.5 to 4 dB apart, and nothing else
		above the floor. That is the 0.6-to-0.4 amplitude ratio of the test message:
		de-emphasis undid the transmitter's pre-emphasis.
		\end{solution}

	\part Type 99.5 into Station. Describe what you hear, and explain why the station
	at 100.3~MHz disappears.
		\begin{solution}[0.9in]
		The audio is a single 600 Hz tone. The Frequency Xlating FIR Filter now shifts 99.5 MHz to
		0 Hz, and the station at 100.3 MHz lands 800 kHz away, far outside the ±90 kHz
		passband.
		\end{solution}

\end{parts}

\question \textbf{AM and FM in the same noise.} Open and run
\texttt{L07-am-vs-fm.grc}. It needs no hardware.

\begin{parts}

	\part Set the deviation to 25~kHz and Input SNR to 30~dB. Record the two output
	SNR meters, and say which output looks cleaner on the frequency sink.
		\begin{solution}[0.8in]
		AM reads about 29.5 dB and FM about 48.4 dB. FM's noise floor sits about 20 dB
		below AM's, so FM is cleaner, although its wider channel lets in more noise.
		\end{solution}

	\part Choose FM under Listen To. Turn the noise up: lower Input SNR slowly until
	you first hear clicks, and record that slider value in the table. Keep going to
	10~dB, and describe what you hear and what the FM meter does. Then repeat the
	click search at 5~kHz and at 50~kHz deviation, and record each FM meter reading
	at 30~dB too.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{1.5in} | K{1.5in} |}
	\hline
	FM deviation & Clicks begin at (dB) & FM meter at 30 dB \\\hline
	5 kHz & \kv{about 10} & \kv{$35.5$} \\\hline
	25 kHz & \kv{about 15 to 18} & \kv{$48.4$} \\\hline
	50 kHz & \kv{about 20 to 23} & \kv{$53.2$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[1.0in]
		Values vary with each listener's ear by a few dB. At 25 kHz the tone is clean at
		20 dB; clicks begin near 15 dB and become frequent at 10 dB, where the tone sounds
		rough and the FM meter has dropped to about the AM meter's 9 dB. The clicks mark the
		point where FM breaks up. AM, in Question 5(b), only grew steadily noisier.
		\end{solution}

\end{parts}

\fullwidth{\large\textbf{Part 3: Writeup} \normalsize(individual, after the lab)}

\question \textbf{Analysis.}

\begin{parts}

	\part Your mono receiver plays a stereo station correctly. Using your multiplex
	screenshot from Question 3(a), explain why.
		\begin{solution}[1.1in]
		The mono audio, the sum of the left and right channels, fills 0 to 15 kHz by
		itself. The pilot, the stereo difference signal, and RDS all sit above 15 kHz, so
		the receiver's 15 kHz low-pass filter removes them and plays the sum.
		\end{solution}

	\part Use your table from Question 6(b) to describe, in words, what a wider FM
	deviation buys and what it costs.
		\begin{solution}[1.3in]
		Wider deviation buys a cleaner output while the signal is strong: the FM meter at
		30 dB climbs from about 35 to about 53 dB as the deviation goes from 5 to 50 kHz.
		It costs bandwidth, and the wider channel lets in more noise, so the clicks start
		sooner as the noise goes up: near 10 dB at 5 kHz but near 20 dB at 50 kHz.
		\end{solution}

	\part Aircraft use AM, and broadcasters use wideband FM. Using your measurements
	from Questions 5 and 6, explain one reason for each choice.
		\begin{solution}[1.2in]
		Broadcast FM: a broadcast station is strong and has a 200 kHz channel, so it
		stays well clear of the clicks, and the wide deviation gives a much cleaner output
		than AM. Aircraft AM: a distant aircraft is weak, and an AM signal degrades gradually
		and stays intelligible where FM would break into clicks. Two aircraft transmitting
		at once are also both heard on AM.
		\end{solution}

\end{parts}
