\vspace{2pt}

This packet is the L\labLesson\ Lab, in three parts, and you turn it in once, complete,
before Lesson 7. Do \textbf{Part 1} on your own before the lab period: predict every
spectral line and receiver effect you will measure. \textbf{Part 2} is your own
measurements, on your own laptop, recorded in class. Do \textbf{Part 3} on your own
after the lab: compare the measurements with the predictions and answer the
analysis questions.

\textbf{Grading.} One submission, graded against Objectives 3.1 and 3.2 on the
course's 1 / 0 scale, with revision allowed under course policy. Paste screenshots
where a question asks for one.

\textbf{No hardware.} Every flowgraph in this lab is a simulation. Nothing transmits,
and you need only your laptop and its speakers or headphones. Start with the volume
low.

The flowgraphs are on the L6 lesson page: \texttt{L06-dsb-am.grc},
\texttt{L06-dsb-sc.grc}, and \texttt{L06-ssb.grc}. All three use the same message,
$x(t) = 0.6\cos(2\pi\cdot 1000\,t) + 0.4\cos(2\pi\cdot 2500\,t)$, on a carrier at
$f_c = 10$ kHz, sampled at 48 kHz.

\question \textbf{Predictions.} You will check each of these at the bench.

\begin{parts}

	\part For DSB-AM with $m = 0.5$ and a carrier of amplitude 1, predict each line's
	frequency and its level relative to the carrier.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.68in} | K{0.68in} | K{0.68in} | K{0.68in} | K{0.68in} |}
	\hline
	Line (kHz) & \kv{$7.5$} & \kv{$9$} & \kv{$10$} & \kv{$11$} & \kv{$12.5$} \\\hline
	Level (dB) & \kv{$-20.0$} & \kv{$-16.5$} & \kv{$0$} & \kv{$-16.5$} & \kv{$-20.0$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.9in]
		Each tone of amplitude $a$ makes sidebands of amplitude $ma/2$ at $f_c \pm f$:
		\eq{ 20\mylog{\dfrac{0.5 \times 0.6}{2}} &= -16.5\db \text{ at 9 and 11 kHz} \\
			20\mylog{\dfrac{0.5 \times 0.4}{2}} &= -20.0\db \text{ at 7.5 and 12.5 kHz} }
		\end{solution}

	\part List the lines you expect for DSB-SC, USB, and LSB.
		\begin{solution}[0.7in]
		DSB-SC: 7.5, 9, 11, and 12.5 kHz, with no line at 10 kHz. USB: 11 and 12.5 kHz.
		LSB: 9 and 7.5 kHz.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the smallest $m$ at which this message overmodulates.
		\begin{solution}[1.0in]
		$m \le 1$ is safe for any message with $\vert x(t)\vert \le 1$, but this message
		is never $-1$: the two cosines never reach $-1$ together, and its minimum is
		$-0.906$, so
		\eq{ m = \dfrac{1}{0.906} = 1.10 }
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		$m = $ \ansbox[1.0in]{$1.10$}
	\end{minipage}

	\part Predict the DSB-SC coherent receiver's output amplitude, relative to its
	value with a matched LO, at each LO phase.

	\begin{center}
	\renewcommand{\arraystretch}{1.25}
	\begin{tabular}{| l | K{0.68in} | K{0.68in} | K{0.68in} | K{0.68in} |}
	\hline
	LO phase & $0^\circ$ & $45^\circ$ & $60^\circ$ & $90^\circ$ \\\hline
	Predicted & \kv{$1$} & \kv{$0.71$} & \kv{$0.5$} & \kv{$0$} \\\hline
	\end{tabular}
	\end{center}

		\begin{solution}[0.4in]
		The output is $x(t)\cos\phi$.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part The USB receiver's LO sits 200 Hz above the carrier. Predict the two
		output tones.
		\begin{solution}[0.7in]
		Each tone moves down by 200 Hz: $1000 - 200 = 800$ Hz and
		$2500 - 200 = 2300$ Hz.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.3in]{800 and 2300 Hz}
	\end{minipage}

\end{parts}

\fullwidth{\large\textbf{Part 3: Writeup} \normalsize(individual, after the lab)}

\question \textbf{Analysis.}

\begin{parts}

	\part Compare your measured DSB-AM levels in Question 2(b) with the predictions in
	Question 1(a), and explain any difference larger than 1 dB.
		\begin{solution}[1.0in]
		They should agree within about 1 dB. A larger difference usually means a line was
		read between FFT bins, or the flowgraph's $m$ or tone amplitudes differ from the
		packet's.
		\end{solution}

	\part Explain in words why an envelope detector recovers DSB-AM but not DSB-SC, and
	what the coherent receiver has that the envelope detector does not.
		\begin{solution}[1.3in]
		DSB-AM's envelope is $1 + m\,x(t)$, which stays positive for $m \le 1$, so it is a
		copy of the message. DSB-SC's envelope is $\vert x(t)\vert$, because the message
		itself goes negative. The coherent receiver has its own copy of the carrier, so
		multiplying by it shifts the signal back down with its sign intact.
		\end{solution}

	\part A DSB-SC LO that is off in phase loses volume, while an SSB receiver that is
	off in frequency changes pitch. Using your measurements, explain which error a
	voice listener tolerates better, and why SSB receivers have a tuning knob instead
	of carrier recovery.
		\begin{solution}[1.3in]
		A small frequency error in SSB moves every tone by the same amount. A listener
		notices the shift at a few tens of hertz, as in Question 5(c), but speech stays
		intelligible out to an error of about 100 Hz. A DSB-SC receiver with a frequency error
		has a phase that drifts continuously, so the output fades in and out as $\cos\phi$
		passes through zero. The ear does not hear phase, so an SSB operator can tune by
		ear, while DSB-SC needs the LO locked to the carrier.
		\end{solution}

\end{parts}
