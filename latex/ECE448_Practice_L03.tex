\question \textbf{Four ways to carry bits.} A link must carry 1 Mb/s.
\begin{parts}

	\part For each family, say what the transmitter changes from one symbol to the
	next: ASK/PAM, PSK, and FSK.
		\begin{solution}[0.9in]
		ASK and PAM change the amplitude, PSK changes the phase of the carrier, and FSK
		changes the frequency.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part How many bits per symbol do 4-PAM, 8-PSK, and 16-PSK carry?
		\begin{solution}[0.8in]
		$\log_2 M$: 4-PAM carries 2, 8-PSK carries 3, and 16-PSK carries 4.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		4-PAM: \ansbox[1.0in]{2 bits}

		\vspace{4pt}
		8-PSK: \ansbox[1.0in]{3 bits}

		\vspace{4pt}
		16-PSK: \ansbox[1.0in]{4 bits}
	\end{minipage}

	\begin{minipage}[t]{0.58\linewidth}
		\part Find the symbol rate the link needs with BPSK and with 8-PSK. This is the
		number a modulator block's samples-per-symbol setting is built around.
		\begin{solution}[1.0in]
		Divide the bit rate by the bits per symbol: BPSK needs
		$10^{6}/1 = 1$ Msym/s, and 8-PSK needs $10^{6}/3 = 333$ ksym/s.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		BPSK: \ansbox[1.3in]{1 Msym/s}

		\vspace{4pt}
		8-PSK: \ansbox[1.3in]{333 ksym/s}
	\end{minipage}

\end{parts}

\question \textbf{Binary PAM and noise.} A binary PAM receiver samples each pulse once.
A 1 is sent as $+1$ V and a 0 as $-1$ V.
\begin{parts}

	\part State the decision rule, and sketch the two points and the threshold on a
	line.
		\begin{solution}[0.9in]
		Decide 1 if the sample is above 0 V and 0 if it is below. The threshold sits at
		0 V, halfway between the two points.
		\end{solution}

	\part Decide each of these received samples: $+0.83$, $-1.20$, $+0.07$, $-0.31$ V.
	The bits sent were 1, 0, 0, 0. Which sample did the receiver decide wrong, and why?
		\begin{solution}[1.0in]
		1, 0, 1, 0. The third sample was sent as $-1$ V, but the noise added more than
		1 V and carried it past the threshold, so the receiver decided it was a 1.
		\end{solution}

	\part The SDR display shows the SNR falling from 14 dB to 4 dB. Describe what
	happens to the received samples, and what you would see on a time sink of the
	slicer's output.
		\begin{solution}[1.1in]
		Each point's spread of samples widens until the spreads meet at 0 V, and some
		samples land on the wrong side. The threshold does not move. On the time sink,
		the slicer's output, which was a clean copy of the sent bits, flips to the
		wrong value from time to time.
		\end{solution}

	\part Name two changes that would keep more samples on the right side of the
	threshold.
		\begin{solution}[0.9in]
		Raise the SNR (more transmit power, a better antenna, a lower-noise first
		stage, a narrower receiver filter), or move the two points farther apart for
		the same power, which antipodal points already do.
		\end{solution}

\end{parts}

\question \textbf{Antipodal, orthogonal, and constellations.}
\begin{parts}

	\part Sketch binary PAM (antipodal) and binary FSK (orthogonal) at the same
	energy, with the boundary between the two points of each. Which pair is
	farther apart?
		\begin{solution}[1.2in]
		Antipodal: points at $-1$ and $+1$ on the I axis, boundary on the Q axis.
		Orthogonal: one point on each axis, at $(1, 0)$ and $(0, 1)$, boundary on the
		diagonal. The antipodal points are farther apart.
		\end{solution}

	\part Both are sent through the same noise. Which one puts more samples in the
	wrong region, and why?
		\begin{solution}[0.9in]
		The orthogonal pair. The clouds are the same size, but its points sit closer to
		their boundary, so more samples stray across.
		\end{solution}

	\part Sketch QPSK and 4-PAM at the same average energy, mark the decision regions
	of each, and explain why a radio uses QPSK rather than 4-PAM on a carrier.
		\begin{solution}[1.3in]
		QPSK has one point per quadrant, and its regions are the quadrants. 4-PAM has
		four points on the I axis, and its regions are strips separated by thresholds
		halfway between neighbors. Both carry two bits, but QPSK uses both axes and
		keeps its points farther apart for the same power, so the same noise puts
		fewer samples in the wrong region.
		\end{solution}

\end{parts}

\question \textbf{Gray coding.} An 8-PSK modem places its points every
$45\mydeg$ around a circle.
\begin{parts}

	\part Label the eight points with a Gray code, starting from 000 at $0\mydeg$.
		\begin{solution}[1.0in]
		Going counterclockwise: 000, 001, 011, 010, 110, 111, 101, 100. Each label
		differs from both of its neighbors, including 100 and 000, in one bit.
		\end{solution}

	\begin{minipage}[t]{0.58\linewidth}
		\part With natural binary labels instead (000, 001, 010, \ldots, 111), how many
		bits does a slip from the point labeled 011 to its neighbor 100 corrupt?
		\begin{solution}[0.7in]
		011 and 100 differ in all three positions, so all three bits are wrong.
		\end{solution}
	\end{minipage}\hfill%
	\begin{minipage}[t]{0.37\linewidth}
		\raggedleft \vspace{12pt}
		\ansbox[1.0in]{3 bits}
	\end{minipage}

	\part Explain in one or two sentences why a Gray code helps, given where noise
	usually pushes a sample.
		\begin{solution}[0.9in]
		Noise almost always pushes a sample into a neighbor's region, not across the
		constellation. With a Gray code every neighbor differs by one bit, so that slip
		costs one bit instead of up to three.
		\end{solution}

\end{parts}

\question \textbf{Choosing a modulation.} In the constellation explorer, QPSK at an
SNR of 12 dB shows almost no samples in the wrong region. Switching to 8-PSK at
the same SNR turns a steady trickle of samples red.
\begin{parts}

	\part Explain what changed on the plot, using the decision regions.
		\begin{solution}[0.9in]
		8-PSK puts twice as many points on the same circle, so they sit closer together
		and the regions are narrower. The clouds are the same size, so they now reach
		the boundaries.
		\end{solution}

	\part What does 8-PSK give you in exchange?
		\begin{solution}[0.7in]
		Three bits per symbol instead of two, so 50\% more data at the same symbol
		rate and bandwidth.
		\end{solution}

	\part A link's SDR display shows only a little more SNR than QPSK needed to stay
	clean. Which modulation do you choose, and what would make you change your
	mind?
		\begin{solution}[1.0in]
		QPSK: its clouds stay inside their regions with the SNR the link has, and the
		margin covers a fade. 8-PSK becomes the choice only if the link gains SNR, from
		more power, better antennas, or a shorter range, enough that its clouds stay
		clean too.
		\end{solution}

\end{parts}

\question \textbf{Reading a constellation.} Three QPSK receivers each show one
problem on a constellation sink. Name the cause of each.
\begin{parts}

	\part The four points sit in the middle of their quadrants, but each is a round
	cloud about a third of the way to the boundaries.
		\begin{solution}[0.6in]
		Noise. The cloud's size shows how good the SNR is; nothing else is wrong.
		\end{solution}

	\part Four tight clouds, each rotated $30\mydeg$ counterclockwise from where it
	belongs, standing still.
		\begin{solution}[0.6in]
		A fixed phase offset between the transmitter and receiver oscillators.
		\end{solution}

	\part The points are smeared into a ring of constant radius.
		\begin{solution}[0.6in]
		A frequency offset: the phase error grows continuously, so the constellation
		rotates through every angle during the capture.
		\end{solution}

	\part Explain why the constellation in part (b) would put every symbol in the
	wrong region if the rotation passed $45\mydeg$, even with no noise.
		\begin{solution}[0.9in]
		The QPSK decision boundaries are the axes, $45\mydeg$ from each point. A
		rotation of $45\mydeg$ puts every point on a boundary, and anything past it puts
		every point in a neighbor's region, so the receiver must correct the phase
		before it decides.
		\end{solution}

\end{parts}
